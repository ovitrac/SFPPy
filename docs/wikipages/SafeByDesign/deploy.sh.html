#!/usr/bin/env bash
# ------------------------------------------------------------------------------
# Deployment script for static index + Surge
# Requirements: bash, python3, surge (logged in), optional: jq (for pretty stats)
#
# default (deflate -9, auto zip name, hide code viewers)
#   ./deploy.sh
# tweak domain + title on the fly
#   DOMAIN="fsbp-v3" TITLE="FSPB v3 – simulation outputs" ./deploy.sh
# add excludes (quote as one string)
#   EXCLUDES=".* .git* node_modules __pycache__ *.map" ./deploy.sh
# try LZMA compression (smaller, slower)
#   ZIP_METHOD=lzma ./deploy.sh
#
#
# Generative Simulation Initiative 🌱 — Dr Olivier Vitrac — contact: olivier.vitrac@gmail.com
# Revision: 2025-09-24

# ------------------------------------------------------------------------------

set -Eeuo pipefail

# --- User config (override via env or edit here) -------------------------------
TITLE=${TITLE:-"Safe Design of Packaging 🌱 with SFPPy 🍏⏩🍎"}
DOMAIN=${DOMAIN:-"SFPPyDesign"}      # surge subdomain (=> <domain>.surge.sh)
ZIPNAME=${ZIPNAME:-"SFPPyDesign.zip"}        # "auto" => <foldername>.zip
ZIP_METHOD=${ZIP_METHOD:-"deflate"}  # deflate|lzma|bzip2
ZIP_LEVEL=${ZIP_LEVEL:-9}            # 0..9 (ignored for lzma)
HTML_OUT=${HTML_OUT:-"files.html"}   # per-folder listing name
EXCLUDES=${EXCLUDES:-""}             # space-separated glob patterns if needed
HIDE_CODE_HTML=${HIDE_CODE_HTML:-1}  # 1=hide generated code viewers from table
EMBED_JSON=${EMBED_JSON:-1}          # 1=embed files.json in HTML (works on file://)
OVERWRITE=${OVERWRITE:-1}            # 1=rewrite all HTML pages
# ------------------------------------------------------------------------------

# --- Helpers -------------------------------------------------------------------
script_dir="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
root_dir="$(pwd)"

info()  { printf "\033[1;36m[i]\033[0m %s\n" "$*"; }
ok()    { printf "\033[1;32m[OK]\033[0m %s\n" "$*"; }
warn()  { printf "\033[1;33m[!]\033[0m %s\n" "$*"; }
err()   { printf "\033[1;31m[✗]\033[0m %s\n" "$*"; }
die()   { err "$*"; exit 1; }

need()  { command -v "$1" >/dev/null 2>&1 || die "Missing dependency: $1"; }

cleanup_html_chains() {
  # clean historical *.html.html* chains (from older versions)
  find . -type f -regex '.*\.html(\.html)+$' -delete 2>/dev/null || true
  find . -type f -name 'files.json.html*' -delete 2>/dev/null || true
}

human() {  # bytes -> human-readable
  local n=$1 u=(B KB MB GB TB) i=0
  while (( n>=1024 && i<${#u[@]}-1 )); do n=$(( (n+512)/1024 )); ((i++)); done
  echo "$n ${u[$i]}"
}

# --- Preamble ------------------------------------------------------------------
echo "Generative Simulation Initiative 🌱 — Dr Olivier Vitrac — contact: olivier.vitrac@gmail.com"
echo "--------------------------------------------------------------------------------------------"

need python3
need surge

[[ -f "${script_dir}/index.py" ]] || die "index.py not found next to this script: ${script_dir}/index.py"

# --- Compose args --------------------------------------------------------------
args=( --html-out "$HTML_OUT" --title "$TITLE" --zip --zip-name "$ZIPNAME"
       --zip-method "$ZIP_METHOD" --zip-level "$ZIP_LEVEL" )

# booleans
(( OVERWRITE ))   && args+=( --overwrite )
(( EMBED_JSON ))  && args+=( --embed-json ) || args+=( --no-embed-json )
(( HIDE_CODE_HTML )) && args+=( --hide-code-html ) || args+=( --no-hide-code-html )

# optional excludes
if [[ -n "$EXCLUDES" ]]; then
  # shellsplit EXCLUDES into array
  read -r -a excl <<<"$EXCLUDES"
  args+=( --exclude "${excl[@]}" )
fi

# --- Run build -----------------------------------------------------------------
info "Cleaning legacy *.html.html chains…"
cleanup_html_chains

info "Building manifest + per-folder pages…"
python3 "${script_dir}/index.py" "${args[@]}"

# --- Show quick stats (if jq available) ---------------------------------------
if command -v jq >/dev/null 2>&1; then
  if [[ -f files.json ]]; then
    files=$(jq -r '.summary.files' files.json)
    folders=$(jq -r '.summary.folders' files.json)
    size=$(jq -r '.summary.total_size' files.json)
    zname=$(jq -r '.summary.zip_name' files.json)
    zhas=$(jq -r '.summary.zip_exists' files.json)
    zc=$(jq -r '.summary.zip_compressed? // 0' files.json)
    zu=$(jq -r '.summary.zip_uncompressed? // 0' files.json)
    ok "Files: $files · Folders: $folders · Total: $(human "$size")"
    if [[ "$zhas" == "true" && "$zname" != "" ]]; then
      ok "Zip: $zname  (compressed $(human "$zc") / uncompressed $(human "$zu"))"
    fi
  fi
else
  warn "jq not found; skipping pretty stats."
fi

# --- Deploy with Surge ---------------------------------------------------------
domain="${DOMAIN}.surge.sh"
info "Deploying to https://${domain} …"
surge ./ "$domain"

ok "Deployed to https://${domain}"

# --- Auto-open in browser (if OPEN=1) -----------------------------------------
if [[ "${OPEN:-1}" -eq 1 ]]; then
  url="https://${domain}"
  info "Opening $url in browser…"
  if command -v xdg-open >/dev/null 2>&1; then
    xdg-open "$url" >/dev/null 2>&1 &
  elif command -v open >/dev/null 2>&1; then
    open "$url"
  elif command -v start >/dev/null 2>&1; then
    start "$url"
  else
    warn "No known open command; please open $url manually."
  fi
fi
